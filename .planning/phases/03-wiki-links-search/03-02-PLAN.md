**Wiki link regex (from research):**
```rust
use regex::Regex;
use std::sync::LazyLock;

// Matches [[title]] or [[title|display text]]
// Uses [^\[\]|]+ to avoid catastrophic backtracking
static WIKI_LINK_REGEX: LazyLock<Regex> = LazyLock::new(|| {
    Regex::new(r"\[\[(?P<title>[^\[\]|]+)(?:\|[^\[\]]*)?\]\]").unwrap()
});
```

**extract_wiki_links function (internal, not a command):**
```rust
pub fn extract_wiki_links(content: &str) -> Vec<String> {
    WIKI_LINK_REGEX
        .captures_iter(content)
        .map(|cap| cap.name("title").unwrap().as_str().trim().to_string())
        .collect()
}
```

**FRONTEND INTEGRATION PATTERN:**

Wiki link extraction is kept as a separate command (not baked into update_note) so the frontend can:
- Debounce calls to avoid extracting on every keystroke
- Show "indexing..." indicator during extraction
- Control timing independently of save

**WHERE to call extractAndSaveWikiLinks:**

In the useAutoSave hook (or similar save mechanism), call extraction AFTER successful save:

```typescript
// Example in useAutoSave.ts or NoteEditor component
import { commands } from '../lib/bindings';
import { useDebouncedCallback } from 'use-debounce';

// Inside your auto-save hook/component:
const extractWikiLinks = useDebouncedCallback(
  async (noteId: number, content: string) => {
    await commands.extractAndSaveWikiLinks(noteId, content);
    // Optionally: queryClient.invalidateQueries(['backlinks']);
  },
  1000 // 1 second debounce after save
);

// In your save handler:
const handleSave = async () => {
  const result = await commands.updateNote(noteId, title, content);
  if (result.status === 'ok') {
    // Trigger wiki link extraction AFTER successful save
    extractWikiLinks(noteId, content);
  }
};
```

Alternatively, in useUpdateNote hook if you're using TanStack Query mutations:

```typescript
// useUpdateNote.ts
export function useUpdateNote() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: async ({ id, title, content }) => {
      const result = await commands.updateNote(id, title, content);
      if (result.status === 'error') throw new Error(result.error);
      return result.data;
    },
    onSuccess: (_, { id, content }) => {
      // Extract wiki links after successful save
      commands.extractAndSaveWikiLinks(id, content);
      // Invalidate backlinks so BacklinksPanel refreshes
      queryClient.invalidateQueries({ queryKey: ['backlinks', id] });
    },
  });
}
```
